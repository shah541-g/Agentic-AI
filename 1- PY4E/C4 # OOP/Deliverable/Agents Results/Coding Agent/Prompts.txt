class Snake {
private:
    std::deque<Point> body;
    Direction currentDir;

public:
    Snake(int startX, int startY) {
        body.push_back({startX, startY});
        body.push_back({startX - 1, startY});
        body.push_back({startX - 2, startY});
        currentDir = Direction::RIGHT;
    }

    const Point& getHead() const {
        return body.front();
    }

    const std::deque<Point>& getBody() const {
        return body;
    }

    void setDirection(Direction dir) {
        currentDir = dir;
    }

    Direction getDirection() const {
        return currentDir;
    }

    void move(bool grow) {
        Point newHead = getHead();
        switch (currentDir) {
            case Direction::LEFT:  newHead.x--; break;
            case Direction::RIGHT: newHead.x++; break;
            case Direction::UP:    newHead.y--; break;
            case Direction::DOWN:  newHead.y++; break;
            case Direction::STOP:  break;
        }
        body.push_front(newHead);
        if (!grow) {
            body.pop_back();
        }
    }

    bool checkSelfCollision() const {
        const Point& head = getHead();
        for (size_t i = 1; i < body.size(); ++i) {
            if (head == body[i]) return true;
        }
        return false;
    }
};

private:
    Point pos;
    int width, height;

public:
    Food(int w, int h) : width(w), height(h) {
        respawn();
    }

    void respawn() {
        pos.x = rand() % (width - 2) + 1;
        pos.y = rand() % (height - 2) + 1;
    }

    const Point& getPosition() const {
        return pos;
    }
};

private:

    int getHeight() const { return height; }

        return p.x <= 0 || p.x >= width - 1 || p.y <= 0 || p.y >= height - 1;
    }
};

                } else {

class GameModel:
    def __init__(self):
        self.snake = [[0, 0], [20, 0], [40, 0]]
        self.direction = "stop"
        self.food = [0, 0]
        self.score = 0
        self.high_score = 0
        self.spawn_food()

    def spawn_food(self):
        x = random.randint(-WINDOW_WIDTH/2 + GRID_SIZE, WINDOW_WIDTH/2 - GRID_SIZE)
        y = random.randint(-WINDOW_HEIGHT/2 + GRID_SIZE, WINDOW_HEIGHT/2 - GRID_SIZE)
        # Align to grid
        self.food = [x - (x % GRID_SIZE), y - (y % GRID_SIZE)]

    def move_snake(self):
        head = list(self.snake[-1])
        if self.direction == 'up':
            head[1] += GRID_SIZE
        elif self.direction == 'down':
            head[1] -= GRID_SIZE
        elif self.direction == 'left':
            head[0] -= GRID_SIZE
        elif self.direction == 'right':
            head[0] += GRID_SIZE

        if self.direction != 'stop':
            self.snake.append(head)
            if not self.check_food_collision():
                self.snake.pop(0)

    def check_food_collision(self):
        if self.snake[-1] == self.food:
            self.score += 10
            if self.score > self.high_score:
                self.high_score = self.score
            self.spawn_food()
            return True
        return False

        head = self.snake[-1]

                head[0] < -WINDOW_WIDTH/2 or 
                head[1] > WINDOW_HEIGHT/2 or
